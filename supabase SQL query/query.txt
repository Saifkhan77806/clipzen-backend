-- =========================================================
-- CLIPZEN E2E DATABASE SCHEMA
-- =========================================================

create extension if not exists pgcrypto;


-- =========================================================
-- PROFILES
-- One application profile per Supabase Auth user.
-- =========================================================

create table public.profiles (
    id uuid primary key
        references auth.users(id)
        on delete cascade,

    display_name text,

    avatar_url text,

    created_at timestamptz not null default now(),

    updated_at timestamptz not null default now()
);


-- =========================================================
-- DEVICES
-- Permanent identity of a CLIPZEN installation.
--
-- devices.id NEVER changes when:
-- - QR is regenerated
-- - pairing link is regenerated
-- - WebSocket reconnects
--
-- It changes only when the device is removed/revoked.
-- =========================================================

create table public.devices (
    id uuid primary key default gen_random_uuid(),

    user_id uuid not null
        references public.profiles(id)
        on delete cascade,

    device_name text not null,

    platform text not null,

    device_identifier text,

    status text not null default 'active',

    last_seen_at timestamptz,

    created_at timestamptz not null default now(),

    updated_at timestamptz not null default now(),

    constraint devices_platform_check
        check (
            platform in (
                'android',
                'windows',
                'macos',
                'linux',
                'ios'
            )
        ),

    constraint devices_status_check
        check (
            status in (
                'active',
                'revoked'
            )
        )
);


create index devices_user_id_idx
    on public.devices(user_id);

create index devices_status_idx
    on public.devices(status);

create index devices_last_seen_at_idx
    on public.devices(last_seen_at);


-- =========================================================
-- DEVICE KEYS
--
-- Public cryptographic identity of a device.
--
-- PRIVATE KEYS ARE NEVER STORED HERE.
-- Private keys remain on the device.
-- =========================================================

create table public.device_keys (
    id uuid primary key default gen_random_uuid(),

    device_id uuid not null
        references public.devices(id)
        on delete cascade,

    key_type text not null,

    public_key text not null,

    key_version integer not null default 1,

    algorithm text not null,

    is_active boolean not null default true,

    created_at timestamptz not null default now(),

    revoked_at timestamptz,

    constraint device_keys_key_type_check
        check (
            key_type in (
                'identity',
                'key_agreement'
            )
        ),

    constraint device_keys_algorithm_check
        check (
            algorithm in (
                'ed25519',
                'x25519'
            )
        )
);


create index device_keys_device_id_idx
    on public.device_keys(device_id);

create index device_keys_active_idx
    on public.device_keys(device_id)
    where is_active = true;


-- =========================================================
-- DEVICE PAIRING TOKENS
--
-- Used by QR codes / connection links.
--
-- IMPORTANT:
-- Raw pairing tokens are NEVER stored.
-- Only their hashes are stored.
--
-- QR regeneration:
-- old token -> revoked
-- new token -> active
-- =========================================================

create table public.device_pairing_tokens (
    id uuid primary key default gen_random_uuid(),

    device_id uuid not null
        references public.devices(id)
        on delete cascade,

    token_hash text not null unique,

    created_at timestamptz not null default now(),

    expires_at timestamptz,

    revoked_at timestamptz,

    last_used_at timestamptz
);


create index device_pairing_tokens_device_id_idx
    on public.device_pairing_tokens(device_id);

create index device_pairing_tokens_active_idx
    on public.device_pairing_tokens(device_id)
    where revoked_at is null;


-- =========================================================
-- DEVICE TRUST RELATIONSHIPS
--
-- Represents an authorized relationship between devices
-- belonging to the same CLIPZEN user.
--
-- The server stores encrypted key material only.
-- It does NOT store usable private keys.
-- =========================================================

create table public.device_trust_relationships (
    id uuid primary key default gen_random_uuid(),

    user_id uuid not null
        references public.profiles(id)
        on delete cascade,

    device_a_id uuid not null
        references public.devices(id)
        on delete cascade,

    device_b_id uuid not null
        references public.devices(id)
        on delete cascade,

    status text not null default 'active',

    established_at timestamptz not null default now(),

    revoked_at timestamptz,

    constraint device_trust_devices_different
        check (
            device_a_id <> device_b_id
        ),

    constraint device_trust_status_check
        check (
            status in (
                'active',
                'revoked'
            )
        ),

    constraint device_trust_unique_pair
        unique (
            device_a_id,
            device_b_id
        )
);


create index device_trust_user_id_idx
    on public.device_trust_relationships(user_id);

create index device_trust_device_a_idx
    on public.device_trust_relationships(device_a_id);

create index device_trust_device_b_idx
    on public.device_trust_relationships(device_b_id);


-- =========================================================
-- CLIPBOARD ITEMS
--
-- SERVER STORES CIPHERTEXT ONLY.
--
-- Plaintext clipboard content MUST NEVER be stored here.
-- =========================================================

create table public.clipboard_items (
    id uuid primary key default gen_random_uuid(),

    user_id uuid not null
        references public.profiles(id)
        on delete cascade,

    source_device_id uuid not null
        references public.devices(id)
        on delete cascade,

    content_type text not null default 'text',

    ciphertext text not null,

    nonce text not null,

    authentication_tag text,

    encryption_algorithm text not null,

    key_version integer not null default 1,

    status text not null default 'pending',

    created_at timestamptz not null default now(),

    accepted_at timestamptz,

    declined_at timestamptz,

    expires_at timestamptz,

    constraint clipboard_content_type_check
        check (
            content_type in (
                'text'
            )
        ),

    constraint clipboard_algorithm_check
        check (
            encryption_algorithm in (
                'xchacha20-poly1305'
            )
        ),

    constraint clipboard_status_check
        check (
            status in (
                'pending',
                'accepted',
                'declined',
                'expired'
            )
        )
);


create index clipboard_items_user_id_idx
    on public.clipboard_items(user_id);

create index clipboard_items_source_device_idx
    on public.clipboard_items(source_device_id);

create index clipboard_items_created_at_idx
    on public.clipboard_items(created_at desc);

create index clipboard_items_status_idx
    on public.clipboard_items(status);


-- =========================================================
-- CLIPBOARD DELIVERIES
--
-- Tracks delivery of encrypted clipboard data to each
-- authorized target device.
-- =========================================================

create table public.clipboard_deliveries (
    id uuid primary key default gen_random_uuid(),

    clipboard_item_id uuid not null
        references public.clipboard_items(id)
        on delete cascade,

    target_device_id uuid not null
        references public.devices(id)
        on delete cascade,

    status text not null default 'pending',

    created_at timestamptz not null default now(),

    delivered_at timestamptz,

    failed_at timestamptz,

    error_message text,

    constraint clipboard_delivery_status_check
        check (
            status in (
                'pending',
                'delivered',
                'failed'
            )
        ),

    constraint clipboard_delivery_unique
        unique (
            clipboard_item_id,
            target_device_id
        )
);


create index clipboard_deliveries_target_device_idx
    on public.clipboard_deliveries(target_device_id);

create index clipboard_deliveries_status_idx
    on public.clipboard_deliveries(status);


-- =========================================================
-- USER SETTINGS
-- =========================================================

create table public.user_settings (
    user_id uuid primary key
        references public.profiles(id)
        on delete cascade,

    clipboard_sync_enabled boolean not null default true,

    notifications_enabled boolean not null default true,

    created_at timestamptz not null default now(),

    updated_at timestamptz not null default now()
);


-- =========================================================
-- UPDATED_AT TRIGGER FUNCTION
-- =========================================================

create or replace function public.set_updated_at()
returns trigger
language plpgsql
as $$
begin
    new.updated_at = now();
    return new;
end;
$$;


create trigger profiles_set_updated_at
before update on public.profiles
for each row
execute function public.set_updated_at();


create trigger devices_set_updated_at
before update on public.devices
for each row
execute function public.set_updated_at();


create trigger user_settings_set_updated_at
before update on public.user_settings
for each row
execute function public.set_updated_at();

create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
    insert into public.profiles (
        id,
        display_name
    )
    values (
        new.id,
        coalesce(
            new.raw_user_meta_data ->> 'full_name',
            new.raw_user_meta_data ->> 'name',
            split_part(new.email, '@', 1)
        )
    );

    insert into public.user_settings (
        user_id
    )
    values (
        new.id
    );

    return new;
end;
$$;


create trigger on_auth_user_created
after insert on auth.users
for each row
execute function public.handle_new_user();
